#[macro_export]
macro_rules! for_each_cfg_field {
    ($cb:ident) => {
        $cb! {
            ( (ShowCpu) (ShowCpu) (Value::Bool(cfg.show_cpu)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.show_cpu = x; true } _ => false } }) { s.push_str(&format!("show_cpu {}\n", self.show_cpu as u32)); } )
            ( (ShowMem) (ShowMem) (Value::Bool(cfg.show_mem)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.show_mem = x; true } _ => false } }) { s.push_str(&format!("show_mem {}\n", self.show_mem as u32)); } )
            ( (ShowGpu) (ShowGpu) (Value::Bool(cfg.show_gpu)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.show_gpu = x; true } _ => false } }) { s.push_str(&format!("show_gpu {}\n", self.show_gpu as u32)); } )
            ( (ShowNet) (ShowNet) (Value::Bool(cfg.show_net)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.show_net = x; true } _ => false } }) { s.push_str(&format!("show_net {}\n", self.show_net as u32)); } )
            ( (ShowTime) (ShowTime) (Value::Bool(cfg.show_time)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.show_time = x; true } _ => false } }) { s.push_str(&format!("show_time {}\n", self.show_time as u32)); } )
            ( (TimeSeconds) (TimeSeconds) (Value::Bool(cfg.time_show_seconds)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.time_show_seconds = x; true } _ => false } }) { s.push_str(&format!("time_seconds {}\n", self.time_show_seconds as u32)); } )
            ( (ShortNames) (ShortNames) (Value::Bool(cfg.short_names)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.short_names = x; true } _ => false } }) { s.push_str(&format!("short_names {}\n", self.short_names as u32)); } )
            ( (MemUnit) (MemUnit) (Value::Select(if cfg.mem_show_gb { 1 } else { 0 })) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { cfg.mem_show_gb = i == 1; true } _ => false } }) { s.push_str(&format!("mem_gb {}\n", self.mem_show_gb as u32)); } )
            ( (FontSize) (FontSize) (Value::Num(cfg.font_size as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => { cfg.font_size = x as f32; true } _ => false } }) { s.push_str(&format!("font_size {:.1}\n", self.font_size)); } )
            ( (FontWeight) (FontWeight) (Value::Select(cfg.font_weight as usize)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { if i <= 2 { cfg.font_weight = i as u8; true } else { false } } _ => false } }) { s.push_str(&format!("font_weight {}\n", self.font_weight)); } )
            ( (MetricGap) (MetricGap) (Value::Num(cfg.metric_gap as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => { cfg.metric_gap = x as u32; true } _ => false } }) { s.push_str(&format!("metric_gap {}\n", self.metric_gap)); } )
            ( (LabelValueGap) (LabelValueGap) (Value::Num(cfg.label_value_gap as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => { cfg.label_value_gap = x as u32; true } _ => false } }) { s.push_str(&format!("label_value_gap {}\n", self.label_value_gap)); } )
            ( (TextShadow) (TextShadow) (Value::Bool(cfg.text_shadow)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.text_shadow = x; true } _ => false } }) { s.push_str(&format!("shadow {}\n", self.text_shadow as u32)); } )
            ( (TextOutline) (TextOutline) (Value::Bool(cfg.text_outline)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.text_outline = x; true } _ => false } }) { s.push_str(&format!("outline {}\n", self.text_outline as u32)); } )
            ( (AutoStart) (AutoStart) (Value::Bool(cfg.autostart)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.autostart = x; true } _ => false } }) { s.push_str(&format!("autostart {}\n", self.autostart as u32)); } )
            ( (DebugLog) (DebugLog) (Value::Bool(cfg.debug_log)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.debug_log = x; true } _ => false } }) { s.push_str(&format!("debug_log {}\n", self.debug_log as u32)); } )
            ( (Topmost) (Topmost) (Value::Bool(cfg.topmost)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.topmost = x; true } _ => false } }) { s.push_str(&format!("topmost {}\n", self.topmost as u32)); } )
            ( (Opacity) (Opacity) (Value::Num(cfg.normal_opacity as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => { cfg.normal_opacity = (x as u32).clamp(0, 100) as u8; true } _ => false } }) { s.push_str(&format!("normal_opacity {}\n", self.normal_opacity)); } )
            ( (NetMode) (NetMode) (Value::Select(match cfg.net_mode { NetMode::Both => 0, NetMode::Down => 1, NetMode::Up => 2 })) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { match i { 0 => { cfg.net_mode = NetMode::Both; true } 1 => { cfg.net_mode = NetMode::Down; true } 2 => { cfg.net_mode = NetMode::Up; true } _ => false } } _ => false } }) { s.push_str(&format!("net_mode {}\n", self.net_mode.to_config())); } )
            ( (Style) (Style) (Value::Select(match cfg.render_style { RenderStyle::Vertical => 1, _ => 0 })) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { match i { 0 => { cfg.render_style = RenderStyle::Strip; true } 1 => { cfg.render_style = RenderStyle::Vertical; true } _ => false } } _ => false } }) { s.push_str(&format!("vert {}\n", self.render_style.to_config())); } )
            ( (RenderEngine) (RenderEngine) (Value::Select(if cfg.render_engine == RenderEngine::D2d { 1 } else { 0 })) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { cfg.render_engine = if i == 1 { RenderEngine::D2d } else { RenderEngine::Gdi }; true } _ => false } }) { s.push_str(&format!("render_engine {}\n", self.render_engine.to_config())); } )
            ( (Lang) (Lang) (Value::Select(cfg.lang as usize)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { match i { 0 => { cfg.lang = Lang::En; true } 1 => { cfg.lang = Lang::Zh; true } _ => false } } _ => false } }) { s.push_str(&format!("lang {}\n", self.lang.to_config())); } )
            ( (PowerEnabled) (PowerEnabled) (Value::Bool(cfg.power_enabled)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.power_enabled = x; true } _ => false } }) { s.push_str(&format!("power {}\n", self.power_enabled as u32)); } )
            ( (PowerQuery) (PowerQuery) (Value::Bool(cfg.power_query)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.power_query = x; true } _ => false } }) { s.push_str(&format!("power_query {}\n", self.power_query as u32)); } )
            ( (GpuPrimaryOnly) (GpuPrimaryOnly) (Value::Bool(cfg.gpu_primary_only)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.gpu_primary_only = x; true } _ => false } }) { s.push_str(&format!("gpu_primary_only {}\n", self.gpu_primary_only as u32)); } )
            ( (Gpu3dOnly) (Gpu3dOnly) (Value::Bool(cfg.gpu_3d_only)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.gpu_3d_only = x; true } _ => false } }) { s.push_str(&format!("gpu_3d_only {}\n", self.gpu_3d_only as u32)); } )
            ( (NetLowFreq) (NetLowFreq) (Value::Bool(cfg.net_low_freq)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.net_low_freq = x; true } _ => false } }) { s.push_str(&format!("net_low_freq {}\n", self.net_low_freq as u32)); } )
            ( (Appearance) (Appearance) (Value::Select(cfg.appearance.to_config() as usize)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Select(i) => { match i { 0 => { cfg.appearance = Appearance::System; true } 1 => { cfg.appearance = Appearance::Light; true } 2 => { cfg.appearance = Appearance::Dark; true } _ => false } } _ => false } }) { s.push_str(&format!("appearance {}\n", self.appearance.to_config())); } )
            ( (ClickThrough) (ClickThrough) (Value::Bool(cfg.click_through)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { cfg.click_through = x; true } _ => false } }) { s.push_str(&format!("click_through {}\n", self.click_through as u32)); } )
            ( (SnapCorner) (SnapCorner) (Value::SelectOpt(if cfg.snap_corner <= 3 { Some(cfg.snap_corner as usize) } else { None })) (|cfg: &mut Config, v: Value| -> bool { match v {
                  Value::SelectOpt(o) => match o { Some(i) if i <= 3 => { cfg.snap_corner = i as u8; true } None => { cfg.snap_corner = SNAP_CORNER_FREE; true } _ => false },
                  Value::Select(i) => { if i <= 3 { cfg.snap_corner = i as u8; true } else { false } }
                  _ => false } }) { s.push_str(&format!("snap_corner {}\n", self.snap_corner)); } )
            ( (MetricShowLabel(usize)) (MetricShowLabel(i)) (Value::Bool(*cfg.metric_show_labels.get(i).unwrap_or(&true))) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { if let Some(slot) = cfg.metric_show_labels.get_mut(i) { *slot = x; true } else { false } } _ => false } }) { } )
            ( (MetricFont(usize)) (MetricFont(i)) (Value::Num(cfg.metric_font_sizes[i].unwrap_or(cfg.font_size) as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => {
                  if i < cfg.metric_font_sizes.len() {
                      let v2 = x as f32;
                      if (v2 - cfg.font_size).abs() < 0.05 { cfg.metric_font_sizes[i] = None; } else { cfg.metric_font_sizes[i] = Some(v2); }
                      true
                  } else { false }
              } _ => false } }) { } )
            ( (MetricOffset(usize, u8)) (MetricOffset(i, c)) (Value::Num(cfg.metric_offsets.get(i).and_then(|o| o.get(c as usize)).copied().unwrap_or(0) as f64)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Num(x) => {
                  match cfg.metric_offsets.get_mut(i).and_then(|o| o.get_mut(c as usize)) {
                      Some(slot) => { *slot = (x.round() as i32).clamp(OFFSET_MIN, OFFSET_MAX); true }
                      None => false,
                  }
              } _ => false } }) { } )
            ( (MetricLabel(usize)) (MetricLabel(i)) (Value::Color(cfg.metric_colors[i].label)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Color(c) => { if i < cfg.metric_colors.len() { cfg.metric_colors[i].label = c; true } else { false } } _ => false } }) { } )
            ( (MetricValue(usize)) (MetricValue(i)) (Value::Color(cfg.metric_colors[i].value)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Color(c) => { if i < cfg.metric_colors.len() { cfg.metric_colors[i].value = c; true } else { false } } _ => false } }) { } )
            ( (MetricValueAuto(usize)) (MetricValueAuto(i)) (Value::Bool(cfg.metric_colors[i].value_auto)) (|cfg: &mut Config, v: Value| -> bool { match v { Value::Bool(x) => { if i < cfg.metric_colors.len() { cfg.metric_colors[i].value_auto = x; true } else { false } } _ => false } }) { } )
        }
    };
}

/// 由 `for_each_cfg_field!` 的变体类型字段 `(Variant)` / `(Variant(T))` 还原出枚举变体声明。
macro_rules! cfg_ref_enum_variant {
    ( ($v:ident) ) => { $v };
    ( ($v:ident ( $($ty:ty),* )) ) => { $v ( $($ty),* ) };
}

/// 生成 `CfgRef` 枚举。
macro_rules! cfg_ref_enum {
    ( $( ( $vty:tt $vpat:tt $get:tt $set:tt $ser:tt ) )* ) => {
        #[derive(Clone, Copy, Debug, PartialEq, Eq)]
        pub enum CfgRef {
            $(
                cfg_ref_enum_variant!($vty),
            )*
        }
    };
}

/// 生成 `CfgRef::get`。
macro_rules! cfg_ref_get {
    ( $( ( $vty:tt $vpat:tt $get:tt $set:tt $ser:tt ) )* ) => {
        impl CfgRef {
            pub fn get(self, cfg: &Config) -> Value {
                match self {
                    $( $vpat => $get, )*
                }
            }
        }
    };
}

/// 生成 `CfgRef::set`（含类型错配 catch-all：debug panic / release 拒绝）。
macro_rules! cfg_ref_set {
    ( $( ( $vty:tt $vpat:tt $get:tt $set:tt $ser:tt ) )* ) => {
        impl CfgRef {
            pub fn set(self, cfg: &mut Config, v: Value) -> bool {
                match (self, v) {
                    $( ($vpat, v) => $set(cfg, v), )*
                    (r, v) => {
                        debug_assert!(false, "CfgRef 类型错配: {r:?} <- {v:?}");
                        return false;
                    }
                }
            }
        }
    };
}
